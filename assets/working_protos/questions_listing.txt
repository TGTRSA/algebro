import { View, Text, ScrollView } from 'react-native';
import { useState, useEffect } from 'react';
import { WebView } from 'react-native-webview';
import rawData from '../assets/questions/algebra/exponents/expansion.json';

interface Equation {
  eq: string;
  level: string;
  technique: string;
  description: string;
  answer: string;
}

const KatexWebView = ({ expression }: { expression: string }) => {
  const html = `
    <!DOCTYPE html>
    <html>
    <head>
      <meta name="viewport" content="width=device-width, initial-scale=1.0">
      <link rel="stylesheet" href="https://cdn.jsdelivr.net/npm/katex@0.16.9/dist/katex.min.css">
      <script src="https://cdn.jsdelivr.net/npm/katex@0.16.9/dist/katex.min.js"></script>
    </head>
    <body style="margin: 0; padding: 12px; background: transparent; display: flex; justify-content: center; align-items: center;">
      <div id="math"></div>
      <script>
        try {
          katex.render("${expression.replace(/\\/g, '\\\\')}", document.getElementById('math'), {
            displayMode: true,
            throwOnError: false
          });
        } catch(err) {
          document.getElementById('math').innerHTML = '<span style="color: red;">Error: ${expression}</span>';
        }
      </script>
    </body>
    </html>
  `;
  
  return <WebView source={{ html }} style={{ height: 80, backgroundColor: 'transparent' }} scrollEnabled={false} />;
};

export default function DebugPage() {
  const [equations, setEquations] = useState<Equation[]>([]);

  useEffect(() => {
    console.log('rawData:', rawData);
    if (rawData && rawData.equations) {
      setEquations(rawData.equations);
    }
  }, []);

  return (
    <ScrollView style={{ padding: 20 }}>
      {equations.map((eq, index) => (
        <View key={index} style={{ marginBottom: 30, padding: 10, borderWidth: 1, borderColor: '#ccc' }}>
          <Text style={{ fontWeight: 'bold' }}>{eq.level} - #{index + 1}</Text>
          <Text style={{ fontStyle: 'italic' }}>{eq.technique}</Text>
          <Text>{eq.description}</Text>
          <View style={{ marginVertical: 10 }}>
            <KatexWebView expression={eq.eq} />
          </View>
          <Text>Answer:</Text>
          <KatexWebView expression={eq.answer} />
        </View>
      ))}
    </ScrollView>
  );
}
